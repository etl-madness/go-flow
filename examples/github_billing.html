go GET_GITHUB_BILLING /Q{}pipeline[1]/Q{}scripts[1]/Q{}script[2] package main import ( "fmt" "io" "net/http" "host/vars" ) func main() { token := vars.GetString("GITHUB_TOKEN") username := vars.GetString("GITHUB_USER") uri := vars.GetString("GITHUB_API_URL") url := fmt.Sprintf("%s/%s/settings/billing/usage", uri, username) req, err := http.NewRequest("GET", url, nil) if err != nil { panic(fmt.Sprintf("Error creating request: %v", err)) } req.Header.Set("Authorization", "Bearer "+token) req.Header.Set("Accept", "application/vnd.github+json") req.Header.Set("X-GitHub-Api-Version", "2022-11-28") client := &http.Client{} resp, err := client.Do(req) if err != nil { panic(fmt.Sprintf("Error executing request: %v", err)) } defer resp.Body.Close() if resp.StatusCode != http.StatusOK { panic(fmt.Sprintf("GitHub API returned non-200 status: %s", resp.Status)) } body, err := io.ReadAll(resp.Body) if err != nil { panic(fmt.Sprintf("Error reading response body: %v", err)) } // Output pure JSON string to stdout so it gets captured by output_var fmt.Print(string(body)) }
